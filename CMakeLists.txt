cmake_minimum_required(VERSION 3.28)

project(crux_pilot LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
option(CRUX_BUILD_HOST_SAMPLE "Build the host Time command-line sample" ON)
option(CRUX_BUILD_REFERENCE_CONTROLLER "Build the early reference controller" ${CRUX_BUILD_HOST_SAMPLE})

add_subdirectory(components/base)
add_subdirectory(components/time)
option(CRUX_BUILD_THREAD "Build the Thread capability" ON)
option(CRUX_BUILD_SYNC "Build the Sync capability" ON)
if(CRUX_BUILD_THREAD)
    add_subdirectory(components/thread)
endif()
if(CRUX_BUILD_SYNC)
    add_subdirectory(components/sync)
endif()
add_subdirectory(components/queue)
add_subdirectory(components/monotonic)
add_subdirectory(components/netif)
add_subdirectory(components/sockets)
add_subdirectory(components/configuration)

add_subdirectory(components/file)
option(CRUX_BUILD_SERIAL "Build the Serial capability" ON)
if(CRUX_BUILD_SERIAL)
    add_subdirectory(components/serial)
endif()

if(CRUX_BUILD_HOST_SAMPLE OR BUILD_TESTING)
    add_library(crux_time_commands STATIC samples/time/src/time_command.c)
    target_include_directories(crux_time_commands
        PUBLIC samples/time/include
        PRIVATE components/time/include
    )
    target_link_libraries(crux_time_commands PUBLIC crux::base)
    target_compile_options(crux_time_commands PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(CRUX_BUILD_HOST_SAMPLE)
    add_subdirectory(samples/time/host)
    add_subdirectory(samples/networking/host)
endif()

if(CRUX_BUILD_REFERENCE_CONTROLLER OR BUILD_TESTING)
    add_library(crux_controller_core STATIC samples/controller/core/controller.c
        samples/controller/core/health.c)
    target_include_directories(crux_controller_core PUBLIC samples/controller/include
        PRIVATE components/time/include)
    target_link_libraries(crux_controller_core PUBLIC crux::base)
    target_compile_options(crux_controller_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(CRUX_BUILD_REFERENCE_CONTROLLER)
    add_subdirectory(samples/controller/host)
endif()

if(BUILD_TESTING)
    enable_language(CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)

    include(FetchContent)
    FetchContent_Declare(
        googletest
        URL https://github.com/google/googletest/archive/b514bdc898e2951020cbdca1304b75f5950d1f59.tar.gz
        URL_HASH SHA256=9257316d65d6259f6596bc4fa5464092305ab3f3d9dc6d7e5780ab462d8baa64
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    )
    set(gtest_force_shared_crt ON CACHE BOOL "Use the shared CRT on Windows" FORCE)
    FetchContent_MakeAvailable(googletest)

    add_subdirectory(tests/time)
    add_subdirectory(tests/controller)
    if(CRUX_BUILD_THREAD AND CRUX_BUILD_SYNC)
        add_subdirectory(tests/portable_execution)
    endif()
    add_subdirectory(tests/queue)
    add_subdirectory(tests/monotonic/host)
    find_package(Threads REQUIRED)
    add_subdirectory(tests/networking)
    add_subdirectory(tests/configuration)

    add_subdirectory(tests/file)
    if(CRUX_BUILD_SERIAL)
        add_subdirectory(tests/serial)
    endif()
    add_executable(crux_posix_probes tests/execution/src/posix.c)
    target_link_libraries(crux_posix_probes PRIVATE Threads::Threads rt)
    target_compile_definitions(crux_posix_probes PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(crux_posix_probes PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME Execution.PosixProbes COMMAND crux_posix_probes)
    set_tests_properties(Execution.PosixProbes PROPERTIES TIMEOUT 10)
endif()

include(cmake/CruxConsumerBoundary.cmake)
if(CRUX_BUILD_SERIAL)
    cmake_language(DEFER CALL crux_assert_portable_usage crux::serial)
endif()
if(CRUX_BUILD_THREAD)
    cmake_language(DEFER CALL crux_assert_portable_usage crux::thread)
endif()
if(CRUX_BUILD_SYNC)
    cmake_language(DEFER CALL crux_assert_portable_usage crux::sync)
endif()
cmake_language(DEFER CALL crux_assert_portable_usage crux::base crux::time crux::sockets crux::netif crux::file)
cmake_language(DEFER CALL crux_assert_portable_usage crux::configuration)
if(TARGET crux_time_commands)
    cmake_language(DEFER CALL crux_assert_portable_consumer crux_time_commands)
endif()
if(TARGET crux_controller_core)
    cmake_language(DEFER CALL crux_assert_portable_consumer crux_controller_core)
endif()
if(TARGET networking_client_fake)
    cmake_language(DEFER CALL crux_assert_portable_consumer networking_client_fake networking_runtime_fake)
endif()
