cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(crux_reference_controller LANGUAGES C)
if(CONFIG_CRUX_M3_MDIO_COMPLETION_FIX)
    include(mdio_completion.cmake)
endif()
target_sources(app PRIVATE ../core/controller.c)
if(CONFIG_CRUX_M3_DEFERRED_OUTPUT OR CONFIG_CRUX_M3_NATIVE_OUTPUT)
    target_sources(app PRIVATE native_logger.c)
endif()
if(CONFIG_CRUX_M3_NETWORK_PROBE AND CONFIG_CRUX_CONTROLLER_M3)
    message(FATAL_ERROR "Select one independent M3 probe composition")
endif()
if(CONFIG_CRUX_M3_NETWORK_PROBE)
    target_sources(app PRIVATE network_probe.c network_client.c)
    if(CONFIG_CRUX_M3_PHY_DIAGNOSTIC)
        target_sources(app PRIVATE network_phy_diagnostic.c)
        if(CONFIG_CRUX_M3_MDIO_FAILURE_TRACE)
            target_sources(app PRIVATE network_mdio_trace.c)
        endif()
    endif()
elseif(CONFIG_CRUX_CONTROLLER_M3)
    target_sources(app PRIVATE m3_main.c device_input.c ../core/health.c)
    if(CONFIG_CRUX_M3_INTEGRATED)
        target_sources(app PRIVATE network_client.c)
    endif()
else()
    target_sources(app PRIVATE main.c)
endif()
if(CONFIG_CRUX_CONTROLLER_M3 OR CONFIG_CRUX_M3_NETWORK_PROBE)
    execute_process(COMMAND git rev-parse --short=12 HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/../../.."
        OUTPUT_VARIABLE crux_m3_source OUTPUT_STRIP_TRAILING_WHITESPACE
        COMMAND_ERROR_IS_FATAL ANY)
    target_compile_definitions(app PRIVATE CRUX_M3_SOURCE="${crux_m3_source}")
endif()

target_include_directories(app PRIVATE ../include)
target_link_libraries(app PRIVATE crux::time)
target_compile_options(app PRIVATE -Wall -Wextra -Werror)
