set(CRUX_SYSTEM_PROVIDER host CACHE STRING "System provider for this composition")
if(NOT CRUX_SYSTEM_PROVIDER STREQUAL "host")
    message(FATAL_ERROR "Crux System unsupported provider: ${CRUX_SYSTEM_PROVIDER}; expected host")
endif()
find_package(Threads REQUIRED)
add_library(crux_system_api INTERFACE)
target_include_directories(crux_system_api INTERFACE include)
target_link_libraries(crux_system_api INTERFACE crux::base)
add_library(crux_system_host STATIC src/system.c ports/linux/system_host.c)
target_include_directories(crux_system_host PRIVATE src)
target_link_libraries(crux_system_host PRIVATE crux_system_api)
if(NOT CMAKE_HAVE_LIBC_PTHREAD)
    target_link_libraries(crux_system_host PRIVATE Threads::Threads)
endif()
target_compile_definitions(crux_system_host PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(crux_system_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_property(TARGET crux_system_host PROPERTY INTERFACE_CRUX_SYSTEM_PROVIDER host)
set_property(TARGET crux_system_host APPEND PROPERTY COMPATIBLE_INTERFACE_STRING CRUX_SYSTEM_PROVIDER)
add_library(crux_system INTERFACE)
add_library(crux::system ALIAS crux_system)
target_link_libraries(crux_system INTERFACE crux_system_api crux_system_host)
