# Private implementation only: deliberately no public alias or include interface.
if(CONFIG_CRUX_PILOT_MODULE)
    add_library(crux_interrupt STATIC ports/zephyr/interrupt_zephyr.c)
    target_link_libraries(crux_interrupt PRIVATE zephyr_interface)
    zephyr_append_cmake_library(crux_interrupt)
    target_compile_options(crux_interrupt PRIVATE -Wall -Wextra -Werror)
else()
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Crux interrupt implementation requires a supported Linux/Zephyr platform")
    endif()
    add_library(crux_interrupt STATIC ports/linux/interrupt_linux.c)
    target_compile_options(crux_interrupt PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
target_include_directories(crux_interrupt PRIVATE src)
