set(CRUX_LOGGING_PROVIDER host CACHE STRING "Logging provider for this composition")
if(NOT CRUX_LOGGING_PROVIDER STREQUAL "host")
    message(FATAL_ERROR "Crux Logging unsupported provider: ${CRUX_LOGGING_PROVIDER}; expected host")
endif()
find_package(Threads REQUIRED)
add_library(crux_logging_api INTERFACE)
target_include_directories(crux_logging_api INTERFACE include)
target_link_libraries(crux_logging_api INTERFACE crux::base)
add_library(crux_logging_host STATIC src/logging.c ports/linux/logging_host.c)
target_include_directories(crux_logging_host PRIVATE src)
target_link_libraries(crux_logging_host PRIVATE crux_logging_api)
if(NOT CMAKE_HAVE_LIBC_PTHREAD)
    target_link_libraries(crux_logging_host PRIVATE Threads::Threads)
endif()
target_compile_definitions(crux_logging_host PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(crux_logging_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_property(TARGET crux_logging_host PROPERTY INTERFACE_CRUX_LOGGING_PROVIDER host)
set_property(TARGET crux_logging_host APPEND PROPERTY COMPATIBLE_INTERFACE_STRING CRUX_LOGGING_PROVIDER)
add_library(crux_logging INTERFACE)
add_library(crux::logging ALIAS crux_logging)
target_link_libraries(crux_logging INTERFACE crux_logging_api crux_logging_host)
