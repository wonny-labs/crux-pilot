get_filename_component(CRUX_FILE_COMPONENT_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(_crux_storage_sample "${CRUX_FILE_COMPONENT_DIR}/../../samples/flash_target_candidate" REALPATH)
get_filename_component(_crux_application "${APPLICATION_SOURCE_DIR}" REALPATH)
set(_crux_rt1060_sample FALSE)
if(CRUX_RT1060_STORAGE_SAMPLE AND _crux_application STREQUAL _crux_storage_sample AND
   CONFIG_BOARD_TARGET STREQUAL "mimxrt1060_evk@B/mimxrt1062/qspi" AND
   CONFIG_FILE_SYSTEM AND CONFIG_FILE_SYSTEM_LITTLEFS)
  set(_crux_rt1060_sample TRUE)
endif()
if(NOT (CONFIG_BOARD_NATIVE_SIM AND CONFIG_FLASH_SIMULATOR) AND NOT _crux_rt1060_sample)
  message(FATAL_ERROR "Crux File requires simulator or exact private RT1060 sample")
endif()
add_library(crux_file_api INTERFACE)
target_include_directories(crux_file_api INTERFACE "${CRUX_FILE_COMPONENT_DIR}/include")
# CPU/ABI/sysroot flags carry no native headers or definitions.
target_compile_options(crux_file_api INTERFACE ${TOOLCHAIN_C_FLAGS})
# Host image width/non-PIE flags must never escape into the ARM usage target.
if(CONFIG_BOARD_NATIVE_SIM)
  target_compile_options(crux_file_api INTERFACE -fno-pic -fno-pie)
  if(CONFIG_64BIT)
    target_compile_options(crux_file_api INTERFACE -m64)
  else()
    target_compile_options(crux_file_api INTERFACE -m32)
  endif()
endif()
add_library(crux_file_zephyr STATIC "${CRUX_FILE_COMPONENT_DIR}/src/file.c" file_native.c)
target_include_directories(crux_file_zephyr PRIVATE "${CRUX_FILE_COMPONENT_DIR}/src" .)
if(NOT _crux_rt1060_sample)
  zephyr_append_cmake_library(crux_file_zephyr)
endif()
target_link_libraries(crux_file_zephyr PRIVATE crux_file_api zephyr_interface)
if(_crux_rt1060_sample)
  target_compile_definitions(crux_file_zephyr PRIVATE CRUX_RT1060_STORAGE_SAMPLE=1)
endif()
target_compile_options(crux_file_zephyr PRIVATE -Wall -Wextra -Werror)
add_library(crux_file INTERFACE)
add_library(crux::file ALIAS crux_file)
target_link_libraries(crux_file INTERFACE crux_file_api)
