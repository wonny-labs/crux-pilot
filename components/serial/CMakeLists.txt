find_package(Threads REQUIRED)
add_library(crux_serial_api INTERFACE)
target_include_directories(crux_serial_api INTERFACE include)
target_link_libraries(crux_serial_api INTERFACE crux::base)
add_library(crux_serial_host STATIC src/serial.c ports/linux/serial_host.c)
target_link_libraries(crux_serial_host PRIVATE crux_serial_api)
# Match current Thread/Sync: supported glibc resolves pthread from libc. Preserve
# a real native link requirement elsewhere so the consumer guard can reject it.
if(NOT CMAKE_HAVE_LIBC_PTHREAD)
    target_link_libraries(crux_serial_host PRIVATE Threads::Threads)
endif()
target_include_directories(crux_serial_host PRIVATE src ports/linux)
target_compile_definitions(crux_serial_host PRIVATE _DEFAULT_SOURCE _POSIX_C_SOURCE=200809L)
target_compile_options(crux_serial_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_property(TARGET crux_serial_host PROPERTY INTERFACE_CRUX_SERIAL_PROVIDER host)
set_property(TARGET crux_serial_host APPEND PROPERTY COMPATIBLE_INTERFACE_STRING CRUX_SERIAL_PROVIDER)
add_library(crux_serial INTERFACE)
add_library(crux::serial ALIAS crux_serial)
target_link_libraries(crux_serial INTERFACE crux_serial_api crux_serial_host)
