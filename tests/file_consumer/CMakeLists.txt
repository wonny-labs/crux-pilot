cmake_minimum_required(VERSION 3.28)
project(crux_file_independent LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CRUX_BUILD_HOST_SAMPLE OFF CACHE BOOL "" FORCE)
set(CRUX_BUILD_REFERENCE_CONTROLLER OFF CACHE BOOL "" FORCE)
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
add_library(file_consumer_c OBJECT consumer.c)
add_library(file_consumer_cpp OBJECT consumer.cpp)
foreach(consumer IN ITEMS file_consumer_c file_consumer_cpp)
    target_link_libraries(${consumer} PRIVATE crux::file)
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    crux_assert_portable_consumer(${consumer})
    add_executable(${consumer}_fixture fixture.c $<TARGET_OBJECTS:${consumer}>)
    target_include_directories(${consumer}_fixture PRIVATE "${CRUX_SOURCE_DIR}/components/file/host")
    target_link_libraries(${consumer}_fixture PRIVATE crux::file)
    target_compile_definitions(${consumer}_fixture PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(${consumer}_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
set_target_properties(file_consumer_c PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
set_target_properties(file_consumer_cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
set_target_properties(file_consumer_c_fixture file_consumer_cpp_fixture PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
enable_testing()
add_test(NAME File.ConsumerC COMMAND file_consumer_c_fixture)
add_test(NAME File.ConsumerCpp COMMAND file_consumer_cpp_fixture)
