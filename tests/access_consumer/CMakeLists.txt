cmake_minimum_required(VERSION 3.28)
project(crux_access_consumer LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
# Selects what scripts/check-access-boundary.py asks this project to do. See that script for each
# probe's expected outcome.
set(ACCESS_PROBE "consumer" CACHE STRING "Boundary probe")
set(CRUX_BUILD_SYNC ON CACHE BOOL "Required by Access")
set(CRUX_BUILD_ACCESS ON CACHE BOOL "Capability under test")
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxAccessBoundary.cmake")
crux_assert_portable_usage(crux::access)
crux_assert_access_adapter_unreachable(crux::access)

# A fake verifier must not exist in a build that a product could use.
if(TARGET crux_access_test_composition)
    message(FATAL_ERROR "the Access test composition must not exist outside Crux's own tests")
endif()

set(adapter_dir "${CRUX_SOURCE_DIR}/components/access/adapter")

if(ACCESS_PROBE STREQUAL "consumer")
    # A project: includes the public header and compiles as C17 and C++17.
    add_library(access_consumer_c OBJECT consumer.c)
    add_library(access_consumer_cpp OBJECT consumer.cpp)
    foreach(target IN ITEMS access_consumer_c access_consumer_cpp)
        target_link_libraries(${target} PRIVATE crux::access)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        crux_assert_portable_consumer(${target})
        crux_assert_access_consumer(${target})
    endforeach()
    set_target_properties(access_consumer_c PROPERTIES C_STANDARD 17 C_EXTENSIONS OFF)
    set_target_properties(access_consumer_cpp PROPERTIES CXX_STANDARD 17 CXX_EXTENSIONS OFF)
elseif(ACCESS_PROBE STREQUAL "private_service")
    # The arrangement crux::web will use: must be accepted, and must not expose the header.
    add_library(access_service STATIC service.c)
    target_link_libraries(access_service PUBLIC crux::access PRIVATE crux_access_adapter)
    add_library(access_service_consumer OBJECT consumer.c)
    target_link_libraries(access_service_consumer PRIVATE access_service)
    crux_assert_access_consumer(access_service_consumer)
elseif(ACCESS_PROBE STREQUAL "private_service_include")
    # The same arrangement, with a project that tries to use the header anyway: must not compile.
    add_library(access_service STATIC service.c)
    target_link_libraries(access_service PUBLIC crux::access PRIVATE crux_access_adapter)
    add_library(access_service_consumer OBJECT include_adapter.c)
    target_link_libraries(access_service_consumer PRIVATE access_service)
    crux_assert_access_consumer(access_service_consumer)
elseif(ACCESS_PROBE STREQUAL "public_service")
    # A service that exposes the adapter: must be rejected at configuration.
    add_library(access_service STATIC service.c)
    target_link_libraries(access_service PUBLIC crux_access_adapter)
    add_library(access_service_consumer OBJECT consumer.c)
    target_link_libraries(access_service_consumer PRIVATE access_service)
    crux_assert_access_consumer(access_service_consumer)
elseif(ACCESS_PROBE STREQUAL "interface_directory")
    # A library that passes the directory on without naming the target: must be rejected.
    add_library(access_copy INTERFACE)
    target_include_directories(access_copy INTERFACE "${adapter_dir}")
    add_library(access_copy_consumer OBJECT consumer.c)
    target_link_libraries(access_copy_consumer PRIVATE access_copy crux::access)
    crux_assert_access_consumer(access_copy_consumer)
elseif(ACCESS_PROBE STREQUAL "direct_directory")
    # A project that adds the directory itself: must be rejected.
    add_library(access_direct_consumer OBJECT consumer.c)
    target_include_directories(access_direct_consumer PRIVATE "${adapter_dir}")
    target_link_libraries(access_direct_consumer PRIVATE crux::access)
    crux_assert_access_consumer(access_direct_consumer)
elseif(ACCESS_PROBE STREQUAL "direct_link")
    # A project that links the internal target itself: must be rejected.
    add_library(access_direct_consumer OBJECT consumer.c)
    target_link_libraries(access_direct_consumer PRIVATE crux_access_adapter)
    crux_assert_access_consumer(access_direct_consumer)
elseif(ACCESS_PROBE STREQUAL "include_adapter")
    # Includes the adapter header through crux::access alone: must not compile.
    add_library(access_include_consumer OBJECT include_adapter.c)
    target_link_libraries(access_include_consumer PRIVATE crux::access)
elseif(ACCESS_PROBE STREQUAL "include_port")
    # Includes the private port header through crux::access alone: must not compile.
    add_library(access_include_consumer OBJECT include_port.c)
    target_link_libraries(access_include_consumer PRIVATE crux::access)
elseif(ACCESS_PROBE STREQUAL "no_provider")
    # A product image with no provider selected: must not link.
    add_executable(access_no_provider no_provider.c)
    target_link_libraries(access_no_provider PRIVATE crux::access)
else()
    message(FATAL_ERROR "unknown ACCESS_PROBE: ${ACCESS_PROBE}")
endif()
