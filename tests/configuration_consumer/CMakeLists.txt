cmake_minimum_required(VERSION 3.28)
project(configuration_independent_consumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CRUX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CRUX_BUILD_HOST_SAMPLE OFF CACHE BOOL "" FORCE)
set(CRUX_BUILD_REFERENCE_CONTROLLER OFF CACHE BOOL "" FORCE)
add_subdirectory("${CRUX_ROOT}" crux)
add_library(configuration_fixture STATIC ../configuration/fault_provider.c)
target_include_directories(configuration_fixture PRIVATE ../configuration
    "${CRUX_ROOT}/components/configuration/include" "${CRUX_ROOT}/components/configuration/src")
target_compile_options(configuration_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(language IN ITEMS c cpp)
    add_library(configuration_${language}_portable OBJECT consumer.${language})
    target_link_libraries(configuration_${language}_portable PRIVATE crux::configuration)
    target_compile_options(configuration_${language}_portable PRIVATE -Wall -Wextra -Wpedantic -Werror)
    crux_assert_portable_consumer(configuration_${language}_portable)
    add_executable(configuration_${language} composition.c)
    target_link_libraries(configuration_${language} PRIVATE configuration_${language}_portable
        crux::configuration configuration_fixture)
endforeach()
