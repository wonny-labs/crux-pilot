cmake_minimum_required(VERSION 3.28)
project(crux_system_consumer LANGUAGES C CXX)
get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CRUX_BUILD_HOST_SAMPLE OFF CACHE BOOL "" FORCE)
set(CRUX_BUILD_REFERENCE_CONTROLLER OFF CACHE BOOL "" FORCE)
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
include(consumers.cmake)
enable_testing()
add_executable(system_consumer host/main.c)
target_include_directories(system_consumer PRIVATE src ../../samples/controller/system ../../samples/controller/include)
target_link_libraries(system_consumer PRIVATE system_runtime system_controller crux::system)
target_compile_options(system_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(system_consumer PROPERTIES C_STANDARD 17 C_EXTENSIONS OFF)
add_test(NAME SystemConsumer.ActualHost COMMAND system_consumer)
add_executable(system_consumer_policy src/contract_test.c src/runtime.c src/consumer.cpp)
target_link_libraries(system_consumer_policy PRIVATE crux_system_api)
target_compile_options(system_consumer_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(system_consumer_policy PROPERTIES C_STANDARD 17 C_EXTENSIONS OFF CXX_STANDARD 17 CXX_EXTENSIONS OFF)
add_test(NAME SystemConsumer.Policy COMMAND system_consumer_policy)
set_tests_properties(SystemConsumer.ActualHost SystemConsumer.Policy PROPERTIES TIMEOUT 10)
