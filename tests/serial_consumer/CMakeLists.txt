cmake_minimum_required(VERSION 3.28)
project(crux_serial_consumer LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(CRUX_BUILD_SERIAL ON CACHE BOOL "Required external capability")
set(CRUX_BUILD_TIME ON CACHE BOOL "Transaction clock")
set(CRUX_TIME_PROVIDER memory CACHE STRING "Controlled development realtime")
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::serial crux::time)
add_library(serial_runtime STATIC src/runtime.c)
add_library(serial_cpp OBJECT src/consumer.cpp)
add_library(serial_controller_portable STATIC "${CRUX_SOURCE_DIR}/samples/serial/controller.c")
foreach(consumer IN ITEMS serial_runtime serial_cpp serial_controller_portable)
    target_link_libraries(${consumer} PRIVATE crux::serial crux::time)
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(${consumer} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF
        CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    crux_assert_portable_consumer(${consumer})
endforeach()
enable_testing()
foreach(kind IN ITEMS c cpp controller controller_stale)
    add_executable(serial_consumer_${kind} host/main.c $<TARGET_OBJECTS:serial_cpp>)
    target_include_directories(serial_consumer_${kind} PRIVATE src "${CRUX_SOURCE_DIR}/samples/serial"
        "${CRUX_SOURCE_DIR}/components/serial/ports/linux")
    target_link_libraries(serial_consumer_${kind} PRIVATE serial_runtime serial_controller_portable
        crux::serial crux::time )
    target_compile_options(serial_consumer_${kind} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME SerialConsumer.${kind} COMMAND serial_consumer_${kind})
    set_tests_properties(SerialConsumer.${kind} PROPERTIES TIMEOUT 5)
endforeach()
target_compile_definitions(serial_consumer_cpp PRIVATE CPP_CONSUMER)
target_compile_definitions(serial_consumer_controller PRIVATE CONTROLLER)
target_compile_definitions(serial_consumer_controller_stale PRIVATE CONTROLLER STALE_ACK)
# Public-symbol doubles verify the independently authored transaction policy.
add_executable(serial_consumer_policy src/contract_test.c src/runtime.c "${CRUX_SOURCE_DIR}/components/time/src/time_types.c")
target_link_libraries(serial_consumer_policy PRIVATE crux::base)
target_include_directories(serial_consumer_policy PRIVATE "${CRUX_SOURCE_DIR}/components/serial/include"
    "${CRUX_SOURCE_DIR}/components/time/include")
target_compile_options(serial_consumer_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(serial_consumer_policy PROPERTIES C_STANDARD 17 C_EXTENSIONS OFF)
add_test(NAME SerialConsumer.Policy COMMAND serial_consumer_policy)
