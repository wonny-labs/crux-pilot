cmake_minimum_required(VERSION 3.28)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(crux_serial_consumers LANGUAGES C CXX)
add_library(serial_runtime STATIC ../src/runtime.c)
add_library(serial_cpp OBJECT ../src/consumer.cpp)
add_library(serial_controller_portable STATIC ../../../samples/serial/controller.c)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::serial crux::monotonic)
foreach(consumer IN ITEMS serial_runtime serial_cpp serial_controller_portable)
    target_link_libraries(${consumer} PRIVATE crux::serial crux::monotonic)
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(${consumer} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF
        CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    crux_assert_portable_consumer(${consumer})
endforeach()
target_sources(app PRIVATE main.c $<TARGET_OBJECTS:serial_cpp>)
target_include_directories(app PRIVATE ../src ../../../samples/serial)
target_link_libraries(app PRIVATE serial_runtime serial_controller_portable crux::serial crux::monotonic)
