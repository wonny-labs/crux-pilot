cmake_minimum_required(VERSION 3.28)
project(crux_monotonic_tests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
get_filename_component(CRUX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET crux::base)
    add_subdirectory("${CRUX_ROOT}/components/base" base)
endif()
add_executable(monotonic_common common_test.c "${CRUX_ROOT}/components/monotonic/src/monotonic.c")
target_include_directories(monotonic_common PRIVATE "${CRUX_ROOT}/components/monotonic/src"
    "${CRUX_ROOT}/components/monotonic/include")
target_link_libraries(monotonic_common PRIVATE crux::base)
target_compile_options(monotonic_common PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME Monotonic.Common COMMAND monotonic_common)
set_tests_properties(Monotonic.Common PROPERTIES TIMEOUT 5)

if(NOT TARGET crux::monotonic)
    add_subdirectory("${CRUX_ROOT}/components/monotonic" monotonic)
endif()
add_executable(monotonic_native native_test.c)
target_link_libraries(monotonic_native PRIVATE crux::monotonic crux_monotonic_host)
target_link_options(monotonic_native PRIVATE -Wl,--wrap=clock_gettime -Wl,--wrap=clock_nanosleep)
target_compile_options(monotonic_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME Monotonic.NativeBoundaries COMMAND monotonic_native)
add_executable(monotonic_real real_test.cpp)
target_link_libraries(monotonic_real PRIVATE crux::monotonic crux_monotonic_host)
target_compile_options(monotonic_real PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME Monotonic.RealHost COMMAND monotonic_real)
set_tests_properties(Monotonic.NativeBoundaries Monotonic.RealHost PROPERTIES TIMEOUT 5)
include("${CRUX_ROOT}/cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::monotonic)
crux_assert_portable_consumer(monotonic_real)
