cmake_minimum_required(VERSION 3.28)
project(crux_monotonic_tests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
get_filename_component(CRUX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET crux::base)
    add_subdirectory("${CRUX_ROOT}/components/base" base)
endif()
add_executable(monotonic_common common_test.c "${CRUX_ROOT}/components/monotonic/src/monotonic.c")
target_include_directories(monotonic_common PRIVATE "${CRUX_ROOT}/components/monotonic/src"
    "${CRUX_ROOT}/components/monotonic/include")
target_link_libraries(monotonic_common PRIVATE crux::base)
target_compile_options(monotonic_common PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME Monotonic.Common COMMAND monotonic_common)
set_tests_properties(Monotonic.Common PROPERTIES TIMEOUT 5)

if(NOT TARGET crux::monotonic)
    add_subdirectory("${CRUX_ROOT}/components/monotonic" monotonic)
endif()
add_executable(monotonic_linux_port port_test.c)
target_link_libraries(monotonic_linux_port PRIVATE crux::monotonic crux_monotonic_linux)
target_link_options(monotonic_linux_port PRIVATE -Wl,--wrap=clock_gettime -Wl,--wrap=clock_nanosleep)
target_compile_options(monotonic_linux_port PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME Monotonic.LinuxPort COMMAND monotonic_linux_port)
# Compile the portable C++ probe against the facade; only the native executable binds Linux.
add_library(monotonic_integration_portable OBJECT integration_test.cpp)
target_link_libraries(monotonic_integration_portable PRIVATE crux::monotonic)
target_compile_options(monotonic_integration_portable PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(monotonic_linux_integration $<TARGET_OBJECTS:monotonic_integration_portable>)
target_link_libraries(monotonic_linux_integration PRIVATE crux::monotonic crux_monotonic_linux)
add_test(NAME Monotonic.LinuxIntegration COMMAND monotonic_linux_integration)
set_tests_properties(Monotonic.LinuxPort Monotonic.LinuxIntegration PROPERTIES TIMEOUT 5)
include("${CRUX_ROOT}/cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::monotonic)
crux_assert_portable_consumer(monotonic_integration_portable)
