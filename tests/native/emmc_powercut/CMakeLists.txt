cmake_minimum_required(VERSION 3.20.0)
get_filename_component(CRUX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
list(APPEND ZEPHYR_EXTRA_MODULES "${CRUX_ROOT}")
include(${CRUX_ROOT}/tests/support/emmc_sim/emmc_sim.cmake)
list(APPEND DTS_ROOT "${CRUX_ROOT}/tests/support/emmc_sim")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(crux_emmc_powercut C)
target_sources(app PRIVATE main.c workloads.c)
if(CONFIG_FAT_FILESYSTEM_ELM)
  target_sources(app PRIVATE fs_fatfs.c)
elseif(CONFIG_CRUX_FS_RELIANCE_EDGE)
  target_sources(app PRIVATE fs_reliance.c)
  target_include_directories(app PRIVATE ${CRUX_ROOT}/zephyr/fs/reliance_edge)
else()
  message(FATAL_ERROR "emmc_powercut needs a file system: build with "
    "-DEXTRA_CONF_FILE=fatfs.conf or -DEXTRA_CONF_FILE=reliance.conf")
endif()
crux_emmc_sim_add(app)
target_compile_options(app PRIVATE -Wall -Wextra -Werror)
