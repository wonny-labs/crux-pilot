cmake_minimum_required(VERSION 3.20.0)
get_filename_component(CRUX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
list(APPEND ZEPHYR_EXTRA_MODULES "${CRUX_ROOT}")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(crux_file_littlefs C)
if(NOT CONFIG_BOARD_NATIVE_SIM OR NOT CONFIG_FLASH_SIMULATOR)
  message(FATAL_ERROR "This fixture operates simulated Flash only")
endif()
set(CRUX ${CMAKE_CURRENT_LIST_DIR}/../../..)
target_sources(app PRIVATE main.c)
target_link_libraries(app PRIVATE crux::file)
target_include_directories(app PRIVATE ${CRUX}/components/file/include
  ${CRUX}/components/file/src ${CRUX}/components/file/ports/zephyr)
target_compile_options(app PRIVATE -Wall -Wextra -Werror)
target_link_options(app PUBLIC -Wl,--wrap=fs_truncate -Wl,--wrap=fs_close -Wl,--wrap=fs_write)

add_library(file_portable_consumer OBJECT consumer.c)
target_link_libraries(file_portable_consumer PRIVATE crux::file)
set_target_properties(file_portable_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_compile_options(file_portable_consumer PRIVATE -Wall -Wextra -Werror)
crux_assert_portable_consumer(file_portable_consumer)
target_sources(app PRIVATE $<TARGET_OBJECTS:file_portable_consumer>)
