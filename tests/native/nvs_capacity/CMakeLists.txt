cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(crux_nvs_capacity_probe C)
if(NOT CONFIG_BOARD_NATIVE_SIM OR NOT CONFIG_FLASH_SIMULATOR)
  message(FATAL_ERROR "This probe is restricted to native_sim simulated Flash")
endif()
target_sources(app PRIVATE main.c records.c)
target_include_directories(app PRIVATE ${ZEPHYR_BASE}/subsys/settings/include)
target_compile_options(app PRIVATE -Wall -Wextra -Werror)
