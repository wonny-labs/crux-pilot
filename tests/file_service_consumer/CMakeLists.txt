cmake_minimum_required(VERSION 3.28)
project(crux_file_service_independent LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
foreach(capability IN ITEMS SYNC FILE ACCESS FILE_SERVICE)
    set(CRUX_BUILD_${capability} ON CACHE BOOL "Required external capability")
endforeach()
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
add_library(file_service_consumer_c OBJECT consumer.c)
add_library(file_service_consumer_cpp OBJECT consumer.cpp)
foreach(consumer IN ITEMS file_service_consumer_c file_service_consumer_cpp)
    target_link_libraries(${consumer} PRIVATE crux::file_service)
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    crux_assert_portable_consumer(${consumer})
endforeach()
set_target_properties(file_service_consumer_c PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON
    C_EXTENSIONS OFF)
set_target_properties(file_service_consumer_cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF)
