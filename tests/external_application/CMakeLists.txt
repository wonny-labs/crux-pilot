cmake_minimum_required(VERSION 3.28)
project(external_application LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
if(PROFILE STREQUAL "time")
    set(CRUX_BUILD_TIME ON CACHE BOOL "")
    set(CRUX_TIME_PROVIDER memory CACHE STRING "")
    set(capabilities time)
elseif(PROFILE STREQUAL "thread")
    set(CRUX_BUILD_THREAD ON CACHE BOOL "")
    set(capabilities thread)
elseif(PROFILE STREQUAL "sync")
    set(CRUX_BUILD_SYNC ON CACHE BOOL "")
    set(capabilities sync)
elseif(PROFILE STREQUAL "messaging")
    set(CRUX_BUILD_QUEUE ON CACHE BOOL "")
    set(CRUX_BUILD_MONOTONIC ON CACHE BOOL "")
    set(capabilities queue monotonic)
elseif(PROFILE STREQUAL "parent")
    set(capabilities base)
else()
    message(FATAL_ERROR "Unknown external test profile")
endif()
add_executable(parent_before parent.c)
add_test(NAME Parent.Before COMMAND parent_before)
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
add_executable(parent_after parent.c)
add_test(NAME Parent.After COMMAND parent_after)
add_executable(parent_cpp parent.cpp)
add_test(NAME Parent.Cpp COMMAND parent_cpp)
if(NOT BUILD_TESTING OR NOT CMAKE_C_STANDARD EQUAL 99 OR NOT CMAKE_CXX_STANDARD EQUAL 20)
    message(FATAL_ERROR "Crux changed parent testing or language configuration")
endif()
foreach(unwanted IN ITEMS gtest gmock crux_time_commands crux_controller_core crux-controller-host)
    if(TARGET ${unwanted})
        message(FATAL_ERROR "Unrequested development target: ${unwanted}")
    endif()
endforeach()
foreach(component IN ITEMS time thread sync queue monotonic netif sockets configuration file)
    if(NOT component IN_LIST capabilities AND TARGET crux::${component})
        message(FATAL_ERROR "Unselected component: ${component}")
    endif()
endforeach()
add_library(portable STATIC consumer.c consumer.cpp)
set_target_properties(portable PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
string(TOUPPER "${PROFILE}" profile_upper)
target_compile_definitions(portable PRIVATE PROFILE_${profile_upper}=1)
foreach(capability IN LISTS capabilities)
    target_link_libraries(portable PRIVATE crux::${capability})
endforeach()
crux_assert_portable_consumer(portable)
add_executable(external_app main.c)
target_link_libraries(external_app PRIVATE portable)
# Final composition declares the used capabilities too, retaining provider compatibility checks.
foreach(capability IN LISTS capabilities)
    target_link_libraries(external_app PRIVATE crux::${capability})
endforeach()
if(PROFILE STREQUAL "messaging" AND NOT OMIT_PROVIDER)
    # Native composition only: portable translation units never receive these providers.
    target_link_libraries(external_app PRIVATE crux::queue_linux crux::monotonic_linux)
endif()
if(CONFLICT_PROVIDER)
    add_library(conflicting_time INTERFACE)
    set_property(TARGET conflicting_time PROPERTY INTERFACE_CRUX_TIME_BACKEND incompatible)
    set_property(TARGET conflicting_time APPEND PROPERTY COMPATIBLE_INTERFACE_STRING CRUX_TIME_BACKEND)
    target_link_libraries(external_app PRIVATE conflicting_time)
endif()
target_link_options(external_app PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/external.map")
add_test(NAME Parent.Application COMMAND external_app)
