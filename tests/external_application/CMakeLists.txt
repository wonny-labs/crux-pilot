cmake_minimum_required(VERSION 3.28)
project(external_application LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
if(PROFILE STREQUAL "time")
    set(CRUX_BUILD_TIME ON CACHE BOOL "")
    set(CRUX_TIME_PROVIDER memory CACHE STRING "")
    set(capabilities time)
elseif(PROFILE STREQUAL "busy_wait")
    set(CRUX_TIME_BUSY_WAIT ON CACHE BOOL "Bounded CPU wait")
    set(CRUX_BUILD_TIME ON CACHE BOOL "Delay's typed-value dependency")
    set(CRUX_TIME_PROVIDER memory CACHE STRING "Explicit development realtime selection")
    set(capabilities time)
elseif(PROFILE STREQUAL "thread")
    set(CRUX_BUILD_THREAD ON CACHE BOOL "")
    set(CRUX_BUILD_TIME ON CACHE BOOL "Thread's clock dependency")
    set(CRUX_TIME_PROVIDER memory CACHE STRING "Controlled development realtime")
    set(capabilities thread time)
elseif(PROFILE STREQUAL "sync")
    set(CRUX_BUILD_SYNC ON CACHE BOOL "")
    set(capabilities sync)
elseif(PROFILE STREQUAL "messaging")
    set(CRUX_BUILD_QUEUE ON CACHE BOOL "")
    set(CRUX_BUILD_TIME ON CACHE BOOL "Typed monotonic clock")
    set(CRUX_TIME_PROVIDER memory CACHE STRING "Explicit development realtime")
    set(CRUX_BUILD_THREAD ON CACHE BOOL "Absolute sleep")
    set(capabilities queue time thread)
elseif(PROFILE STREQUAL "serial")
    set(CRUX_BUILD_SERIAL ON CACHE BOOL "")
    set(CRUX_SERIAL_PROVIDER host CACHE STRING "")
    set(capabilities serial)
elseif(PROFILE STREQUAL "logging")
    set(CRUX_BUILD_LOGGING ON CACHE BOOL "")
    set(CRUX_LOGGING_PROVIDER host CACHE STRING "")
    set(capabilities logging)
elseif(PROFILE STREQUAL "system")
    set(CRUX_BUILD_SYSTEM ON CACHE BOOL "")
    set(CRUX_SYSTEM_PROVIDER host CACHE STRING "")
    set(capabilities system)
elseif(PROFILE STREQUAL "parent")
    set(capabilities base)
else()
    message(FATAL_ERROR "Unknown external test profile")
endif()
add_executable(parent_before parent.c)
add_test(NAME Parent.Before COMMAND parent_before)
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
add_executable(parent_after parent.c)
add_test(NAME Parent.After COMMAND parent_after)
add_executable(parent_cpp parent.cpp)
add_test(NAME Parent.Cpp COMMAND parent_cpp)
if(NOT BUILD_TESTING OR NOT CMAKE_C_STANDARD EQUAL 99 OR NOT CMAKE_CXX_STANDARD EQUAL 20)
    message(FATAL_ERROR "Crux changed parent testing or language configuration")
endif()
foreach(unwanted IN ITEMS gtest gmock crux_time_commands crux_controller_core crux-controller-host)
    if(TARGET ${unwanted})
        message(FATAL_ERROR "Unrequested development target: ${unwanted}")
    endif()
endforeach()
foreach(component IN ITEMS time thread sync queue netif sockets configuration file serial logging system)
    if(NOT component IN_LIST capabilities AND TARGET crux::${component})
        message(FATAL_ERROR "Unselected component: ${component}")
    endif()
endforeach()
add_library(portable STATIC consumer.c consumer.cpp)
set_target_properties(portable PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
string(TOUPPER "${PROFILE}" profile_upper)
target_compile_definitions(portable PRIVATE PROFILE_${profile_upper}=1)
foreach(capability IN LISTS capabilities)
    target_link_libraries(portable PRIVATE crux::${capability})
endforeach()
crux_assert_portable_consumer(portable)
add_executable(external_app main.c)
target_link_libraries(external_app PRIVATE portable)
# Final composition declares the used capabilities too, retaining provider compatibility checks.
foreach(capability IN LISTS capabilities)
    target_link_libraries(external_app PRIVATE crux::${capability})
endforeach()
if(PROFILE STREQUAL "messaging" AND NOT OMIT_PROVIDER)
    # Native composition only: portable translation units never receive these providers.
    target_link_libraries(external_app PRIVATE crux::queue_linux )
endif()
if(CONFLICT_PROVIDER AND PROFILE STREQUAL "serial")
    add_library(conflicting_serial INTERFACE)
    set_property(TARGET conflicting_serial PROPERTY INTERFACE_CRUX_SERIAL_PROVIDER zephyr)
    set_property(TARGET conflicting_serial APPEND PROPERTY COMPATIBLE_INTERFACE_STRING CRUX_SERIAL_PROVIDER)
    target_link_libraries(external_app PRIVATE conflicting_serial)
elseif(CONFLICT_PROVIDER AND PROFILE STREQUAL "logging")
    add_library(conflicting_logging INTERFACE)
    set_property(TARGET conflicting_logging PROPERTY INTERFACE_CRUX_LOGGING_PROVIDER zephyr)
    set_property(TARGET conflicting_logging APPEND PROPERTY COMPATIBLE_INTERFACE_STRING CRUX_LOGGING_PROVIDER)
    target_link_libraries(external_app PRIVATE conflicting_logging)
elseif(CONFLICT_PROVIDER AND PROFILE STREQUAL "system")
    add_library(conflicting_system INTERFACE)
    set_property(TARGET conflicting_system PROPERTY INTERFACE_CRUX_SYSTEM_PROVIDER zephyr)
    set_property(TARGET conflicting_system APPEND PROPERTY COMPATIBLE_INTERFACE_STRING CRUX_SYSTEM_PROVIDER)
    target_link_libraries(external_app PRIVATE conflicting_system)
elseif(CONFLICT_PROVIDER)
    add_library(conflicting_time INTERFACE)
    set_property(TARGET conflicting_time PROPERTY INTERFACE_CRUX_TIME_BACKEND incompatible)
    set_property(TARGET conflicting_time APPEND PROPERTY COMPATIBLE_INTERFACE_STRING CRUX_TIME_BACKEND)
    target_link_libraries(external_app PRIVATE conflicting_time)
endif()
target_link_options(external_app PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/external.map")
if(PROFILE STREQUAL "time" AND CRUX_TIME_PROVIDER STREQUAL "linux")
    target_compile_definitions(portable PRIVATE PROFILE_TIME_LINUX=1)
    target_sources(external_app PRIVATE realtime_write_guard.c)
    target_compile_definitions(external_app PRIVATE _POSIX_C_SOURCE=200809L)
    target_link_options(external_app PRIVATE -Wl,--wrap=clock_settime)
endif()
add_test(NAME Parent.Application COMMAND external_app)
