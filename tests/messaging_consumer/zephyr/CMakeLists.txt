cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(crux_messaging_runtime_native LANGUAGES C)
add_library(messaging_portable STATIC ../runtime.c)
target_link_libraries(messaging_portable PRIVATE crux::queue crux::monotonic crux::execution)
set_target_properties(messaging_portable PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
target_compile_options(messaging_portable PRIVATE -Wall -Wextra -Wpedantic -Werror)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::queue crux::monotonic crux::execution)
crux_assert_portable_consumer(messaging_portable)
target_sources(app PRIVATE main.c)
target_link_libraries(app PRIVATE messaging_portable crux::queue crux::monotonic crux::execution)
