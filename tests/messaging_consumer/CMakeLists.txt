cmake_minimum_required(VERSION 3.28)
project(crux_messaging_consumer LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(CRUX_BUILD_QUEUE ON CACHE BOOL "Required external capability")
set(CRUX_BUILD_MONOTONIC ON CACHE BOOL "Required external capability")
set(CRUX_BUILD_THREAD ON CACHE BOOL "Required external capability")
set(CRUX_BUILD_SYNC ON CACHE BOOL "Required external capability")
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::queue crux::monotonic crux::thread crux::sync)
add_library(messaging_runtime STATIC runtime.c)
target_link_libraries(messaging_runtime PRIVATE crux::queue crux::monotonic crux::thread crux::sync)
add_library(messaging_cpp OBJECT main.cpp)
target_link_libraries(messaging_cpp PRIVATE crux::queue crux::monotonic crux::thread crux::sync)
add_library(messaging_controller_portable OBJECT "${CRUX_SOURCE_DIR}/samples/messaging/controller.c")
target_link_libraries(messaging_controller_portable PRIVATE crux::monotonic)
# Only the composition selects native providers; no provider usage is exposed to
# the independently compiled C17/C++17 portable source targets above.
add_executable(messaging_consumer $<TARGET_OBJECTS:messaging_cpp>)
target_link_libraries(messaging_consumer PRIVATE messaging_runtime crux::queue crux::monotonic
    crux::thread crux::sync crux::queue_host crux::monotonic_host)
add_executable(messaging_controller $<TARGET_OBJECTS:messaging_controller_portable>)
target_link_libraries(messaging_controller PRIVATE crux::monotonic crux::monotonic_host)
foreach(consumer IN ITEMS messaging_runtime messaging_cpp messaging_controller_portable)
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(${consumer} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES
        C_EXTENSIONS OFF CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    crux_assert_portable_consumer(${consumer})
endforeach()
enable_testing()
add_test(NAME Messaging.RuntimeConsumer COMMAND messaging_consumer)
add_test(NAME Messaging.ControllerConsumer COMMAND messaging_controller)
set_tests_properties(Messaging.RuntimeConsumer Messaging.ControllerConsumer PROPERTIES TIMEOUT 20)
