cmake_minimum_required(VERSION 3.28)
project(crux_component_consumer LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
if(NOT CRUX_COMPONENT MATCHES "^(thread|sync)$")
    message(FATAL_ERROR "Select CRUX_COMPONENT=thread or sync")
endif()
string(TOUPPER "${CRUX_COMPONENT}" component_upper)
set(CRUX_BUILD_${component_upper} ON CACHE BOOL "Selected standalone capability")
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
if(CRUX_COMPONENT STREQUAL "thread")
    set(other sync)
else()
    set(other thread)
endif()
if(TARGET crux::${other} OR TARGET crux_${other}_host OR TARGET crux::execution)
    message(FATAL_ERROR "Unselected component is present")
endif()
add_library(component_portable STATIC consumer.c consumer.cpp)
add_executable(component_consumer runner.c)
if(CRUX_COMPONENT STREQUAL "sync")
    find_package(Threads REQUIRED)
    target_sources(component_consumer PRIVATE native_worker.c)
    target_link_libraries(component_consumer PRIVATE Threads::Threads)
endif()
string(TOUPPER "${CRUX_COMPONENT}" component_upper)
foreach(consumer IN ITEMS component_portable component_consumer)
    target_link_libraries(${consumer} PRIVATE crux::${CRUX_COMPONENT})
    target_compile_definitions(${consumer} PRIVATE CRUX_TEST_${component_upper}=1)
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(${consumer} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES
        C_EXTENSIONS OFF CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
endforeach()
target_link_libraries(component_consumer PRIVATE component_portable)
target_link_options(component_consumer PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/consumer.map")
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_consumer(component_portable)
enable_testing()
add_test(NAME Component.${CRUX_COMPONENT}Only COMMAND component_consumer)
set_tests_properties(Component.${CRUX_COMPONENT}Only PROPERTIES TIMEOUT 10)
