cmake_minimum_required(VERSION 3.20)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(crux_component_consumer LANGUAGES C CXX)
if(CONFIG_CRUX_THREAD AND NOT CONFIG_CRUX_SYNC)
    set(component thread)
    set(other sync)
elseif(CONFIG_CRUX_SYNC AND NOT CONFIG_CRUX_THREAD)
    set(component sync)
    set(other thread)
else()
    message(FATAL_ERROR "Select exactly one Thread/Sync component")
endif()
if(TARGET crux::${other} OR TARGET crux_${other}_zephyr OR TARGET crux::execution)
    message(FATAL_ERROR "Unselected component is present")
endif()
string(TOUPPER "${component}" component_upper)
add_library(component_portable STATIC ../consumer.c ../consumer.cpp)
target_link_libraries(component_portable PRIVATE crux::${component})
target_compile_definitions(component_portable PRIVATE CRUX_TEST_${component_upper}=1)
target_compile_options(component_portable PRIVATE -Wall -Wextra -Werror)
set_target_properties(component_portable PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES
    C_EXTENSIONS OFF CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
target_sources(app PRIVATE ../runner.c)
target_link_libraries(app PRIVATE component_portable crux::${component})
if(component STREQUAL "sync")
    target_sources(app PRIVATE native_worker.c)
endif()
target_compile_options(app PRIVATE -Wall -Wextra -Werror)
crux_assert_portable_consumer(component_portable)
