# Application image for the functional level. Crux arrives only as EXTRA_ZEPHYR_MODULES.
cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(consumer_poc_app LANGUAGES C)
target_sources(app PRIVATE src/main.c ../src/app_run.c ../src/periodic.c)
target_include_directories(app PRIVATE ../include)
target_link_libraries(app PRIVATE crux::time crux::thread)
target_compile_options(app PRIVATE -Wall -Wextra -Werror)
if(CONSUMER_DEFECT_REBASE)
    target_compile_definitions(app PRIVATE CONSUMER_DEFECT_REBASE)
endif()
