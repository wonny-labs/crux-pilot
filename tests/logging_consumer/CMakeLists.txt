cmake_minimum_required(VERSION 3.28)
project(crux_logging_consumer LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CRUX_BUILD_LOGGING ON CACHE BOOL "Required external capability")
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::logging)
add_library(logging_runtime STATIC src/runtime.c src/consumer.cpp)
add_library(logging_controller STATIC "${CRUX_SOURCE_DIR}/samples/logging/controller.c")
foreach(consumer IN ITEMS logging_runtime logging_controller)
    target_link_libraries(${consumer} PRIVATE crux::logging)
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(${consumer} PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF
        CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    crux_assert_portable_consumer(${consumer})
endforeach()
enable_testing()
add_executable(logging_consumer host/main.c)
target_include_directories(logging_consumer PRIVATE src "${CRUX_SOURCE_DIR}/samples/logging")
target_link_libraries(logging_consumer PRIVATE logging_runtime logging_controller crux::logging)
set_target_properties(logging_consumer PROPERTIES C_STANDARD 17 C_EXTENSIONS OFF)
add_test(NAME LoggingConsumer.ActualHost COMMAND logging_consumer)
add_executable(logging_consumer_policy src/contract_test.c src/runtime.c src/consumer.cpp)
target_link_libraries(logging_consumer_policy PRIVATE crux_logging_api)
target_compile_options(logging_consumer_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(logging_consumer_policy PROPERTIES C_STANDARD 17 C_EXTENSIONS OFF CXX_STANDARD 17 CXX_EXTENSIONS OFF)
add_test(NAME LoggingConsumer.Policy COMMAND logging_consumer_policy)
