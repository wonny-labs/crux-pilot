cmake_minimum_required(VERSION 3.28)
project(crux_independent_consumer LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(CRUX_BUILD_TIME ON CACHE BOOL "Required external capability")
set(CRUX_TIME_PROVIDER memory CACHE STRING "Development/test Time implementation")
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::base crux::time)
add_executable(consumer_c consumer.c)
add_executable(consumer_cpp consumer.cpp)
foreach(consumer IN ITEMS consumer_c consumer_cpp)
    target_link_libraries(${consumer} PRIVATE crux::base crux::time)
    crux_assert_portable_consumer(${consumer})
    target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
set_target_properties(consumer_c PROPERTIES C_STANDARD 17 C_EXTENSIONS OFF)
set_target_properties(consumer_cpp PROPERTIES CXX_STANDARD 17 CXX_EXTENSIONS OFF)
