cmake_minimum_required(VERSION 3.28)
project(crux_execution_runtime_consumer LANGUAGES C)

if(NOT DEFINED CRUX_SOURCE_DIR)
    message(FATAL_ERROR "Supply -DCRUX_SOURCE_DIR=/absolute/path/to/crux")
endif()
set(CRUX_BUILD_THREAD ON CACHE BOOL "Required external capability")
set(CRUX_BUILD_SYNC ON CACHE BOOL "Required external capability")
add_subdirectory("${CRUX_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/crux")

add_executable(crux_runtime_consumer src/main.c src/runtime_fixture.c)
target_include_directories(crux_runtime_consumer PRIVATE include)
target_compile_features(crux_runtime_consumer PRIVATE c_std_17)
set_target_properties(crux_runtime_consumer PROPERTIES C_EXTENSIONS OFF)
target_compile_options(crux_runtime_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(crux_runtime_consumer PRIVATE crux::thread crux::sync)
enable_testing()
add_test(NAME Execution.RuntimeConsumer COMMAND crux_runtime_consumer)
set_tests_properties(Execution.RuntimeConsumer PROPERTIES TIMEOUT 60)
