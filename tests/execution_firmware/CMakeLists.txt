cmake_minimum_required(VERSION 3.28)
project(crux_firmware_consumer LANGUAGES C CXX)
if(NOT CRUX_SOURCE_DIR)
    message(FATAL_ERROR "Supply CRUX_SOURCE_DIR")
endif()
set(CRUX_BUILD_THREAD ON CACHE BOOL "Required external capability")
set(CRUX_BUILD_SYNC ON CACHE BOOL "Required external capability")
add_subdirectory("${CRUX_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/crux")
add_library(crux_firmware_portable STATIC src/consumer.c ../portable_execution/public_probe.c)
target_link_libraries(crux_firmware_portable PRIVATE crux::thread crux::sync)
target_compile_features(crux_firmware_portable PRIVATE c_std_17)
set_target_properties(crux_firmware_portable PROPERTIES C_EXTENSIONS OFF)
target_compile_options(crux_firmware_portable PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(crux_firmware_consumer src/main.cpp)
target_compile_features(crux_firmware_consumer PRIVATE cxx_std_17)
target_compile_options(crux_firmware_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(crux_firmware_consumer PRIVATE crux_firmware_portable crux::thread crux::sync)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/consumer-boundary.json"
    CONTENT "{\"links\":\"$<TARGET_PROPERTY:crux_firmware_portable,LINK_LIBRARIES>\"}\n")
enable_testing()
add_test(NAME Execution.FirmwareConsumer COMMAND crux_firmware_consumer)
set_tests_properties(Execution.FirmwareConsumer PROPERTIES TIMEOUT 20)
