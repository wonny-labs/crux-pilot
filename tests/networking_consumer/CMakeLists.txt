cmake_minimum_required(VERSION 3.28)
project(independent_networking_consumer LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CRUX_SOURCE)
    message(FATAL_ERROR "Set CRUX_SOURCE to the SDK source tree")
endif()
set(CRUX_BUILD_SOCKETS ON CACHE BOOL "Required external capability")
set(CRUX_BUILD_NETIF ON CACHE BOOL "Required external capability")
add_subdirectory("${CRUX_SOURCE}" sdk)
foreach(language c cpp)
    add_executable(consumer_${language} main.${language})
    target_link_libraries(consumer_${language} PRIVATE crux::sockets crux::netif)
    target_compile_options(consumer_${language} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    crux_assert_portable_consumer(consumer_${language})
endforeach()
# These are the public usage requirements, not the native composition environment.
get_target_property(netif_includes crux_netif INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(netif_links crux_netif INTERFACE_LINK_LIBRARIES)
file(WRITE "${CMAKE_BINARY_DIR}/public-requirements.txt" "${netif_includes}\n${netif_links}\n")
